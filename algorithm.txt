    اطلب من المستخدم إدخال درجة الطالب أو كتابة "خروج"
    
    إذا كانت القيمة = "خروج"
        اطبع "تم إنهاء البرنامج"
        توقف
    
    حاول تحويل الإدخال إلى رقم
    إذا فشل التحويل
        اطبع "إدخال غير صحيح"
        استمر
    
    إذا الدرجة >= 90
        التقدير = "A"
    وإلا إذا >= 80
        التقدير = "B"
    وإلا إذا >= 70
        التقدير = "C"
    وإلا إذا >= 60
        التقدير = "D"
    وإلا
        التقدير = "F"
    
    اطبع التقدير
